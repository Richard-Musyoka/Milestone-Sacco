    private const string Green = "#0B7A52", Blue = "#2F6FEB", Gold = "#C99A2E", Red = "#D64545";

    private async Task Pick(string value)
    {
        model.ReportType = value;
        await LoadReportData();
    }

    private async Task Preset(int months)
    {
        model.EndDate = DateTime.Today;
        model.StartDate = DateTime.Today.AddMonths(-months);
        await LoadReportData();
    }

    private static string Money(decimal v) => "KES " + v.ToString("N0", CultureInfo.InvariantCulture);

    private string Sub => ReportTitle + " · " + model.StartDate.ToString("d MMM yyyy", CultureInfo.InvariantCulture) + " – " + model.EndDate.ToString("d MMM yyyy", CultureInfo.InvariantCulture);
